// Fonction pour générer une fausse barre de progression
function obtenirBarreProgression(actuel, total) {
    const tailleBarre = 15;
    const progression = Math.round((actuel / total) * tailleBarre);
    const vide = tailleBarre - progression;
    return `[${"█".repeat(progression)}${"░".repeat(vide)}]`;
}

// Fonction pour obtenir un délai aléatoire (entre 50ms et 500ms)
function obtenirDelaiAleatoire() {
    // Mélange de délais : parfois rapide, parfois très lent
    const types = [
        () => Math.floor(Math.random() * 50) + 20,      // Très rapide (20-70ms)
        () => Math.floor(Math.random() * 100) + 50,     // Rapide (50-150ms)
        () => Math.floor(Math.random() * 200) + 100,    // Moyen (100-300ms)
        () => Math.floor(Math.random() * 400) + 200,    // Lent (200-600ms)
        () => Math.floor(Math.random() * 800) + 500,    // Très lent (500-1300ms)
        () => Math.floor(Math.random() * 1500) + 1000   // Extrêmement lent (1000-2500ms)
    ];
    
    // Poids pour donner plus de chances aux délais moyens
    const poids = [1, 2, 3, 2, 1, 1];
    const totalPoids = poids.reduce((a, b) => a + b, 0);
    let r = Math.random() * totalPoids;
    
    for (let i = 0; i < poids.length; i++) {
        r -= poids[i];
        if (r <= 0) {
            return types[i]();
        }
    }
    return types[2]();
}

// Fonction pour générer un "effet de charge" avec des points de suspension
function obtenirEffetCharge(compteur) {
    const points = ["", ".", "..", "...", "....", "....."];
    const index = compteur % points.length;
    return points[index];
}

function genererLogBizarre() {
    if (compteur >= LIMITE) {
        console.log(`\n🎉 CHARGEMENT TERMINÉ ! ${LIMITE} anomalies chargées avec succès.`);
        clearInterval(intervalle);
        return;
    }

    const ico = icones[Math.floor(Math.random() * icones.length)];
    const stat = status[Math.floor(Math.random() * status.length)];
    const msg = messages[Math.floor(Math.random() * messages.length)];
    
    const date = new Date().toISOString().replace('T', ' ').substring(0, 19);
    const barre = obtenirBarreProgression(compteur, LIMITE);
    const effet = obtenirEffetCharge(compteur);

    // Ajout d'un pourcentage pour plus de réalisme
    const pourcentage = Math.round((compteur / LIMITE) * 100);

    // Ligne avec effet de chargement
    const ligneTerminal = `${ico} [${date}] [${stat}] ${barre} ${pourcentage}%${effet} (${compteur}/${LIMITE}) -> ${msg}\n`;

    // Ligne simplifiée pour le fichier
    const ligneFichier = `[${date}] [${stat}] ${pourcentage}% (${compteur}/${LIMITE}) -> ${msg}\n`;

    process.stdout.write(ligneTerminal);
    fs.appendFileSync('matrix_anomalies_slow.log', ligneFichier, 'utf8');

    compteur++;

    // Recalcul du délai pour la prochaine itération
    clearInterval(intervalle);
    const nouveauDelai = obtenirDelaiAleatoire();
    
    // Affiche la vitesse actuelle de temps en temps (pour le fun)
    if (compteur % 20 === 0 && compteur > 0) {
        process.stdout.write(`🐌 Vitesse de chargement : ${nouveauDelai}ms par ligne\n`);
    }
    
    intervalle = setInterval(genererLogBizarre, nouveauDelai);
}
